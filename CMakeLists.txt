cmake_minimum_required(VERSION 3.24)
project(dvc VERSION 0.1.0 LANGUAGES C CXX)
include(CTest)
include(GNUInstallDirs)
option(DVC_ENABLE_DIRECTML "Enable the Windows DirectML execution provider" OFF)
option(DVC_BUILD_EXAMPLES "Build the WAV converter" ON)
set(ONNXRUNTIME_ROOT "" CACHE PATH "ONNX Runtime SDK with include/ and lib/")
find_path(ONNXRUNTIME_INCLUDE_DIR onnxruntime_cxx_api.h HINTS "${ONNXRUNTIME_ROOT}/include" REQUIRED)
find_library(ONNXRUNTIME_LIBRARY NAMES onnxruntime HINTS "${ONNXRUNTIME_ROOT}/lib" REQUIRED)
add_library(dvc SHARED src/dvc.cpp)
add_library(dvc::dvc ALIAS dvc)
target_compile_features(dvc PRIVATE cxx_std_17)
target_compile_definitions(dvc PRIVATE DVC_BUILDING_LIBRARY)
if(WIN32)
  target_compile_definitions(dvc PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
target_include_directories(dvc PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_include_directories(dvc PRIVATE "${ONNXRUNTIME_INCLUDE_DIR}")
target_link_libraries(dvc PRIVATE "${ONNXRUNTIME_LIBRARY}")
set_target_properties(dvc PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES VERSION ${PROJECT_VERSION} SOVERSION 0)
if(MSVC)
  target_compile_options(dvc PRIVATE /W4 /utf-8 /EHsc)
else()
  target_compile_options(dvc PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(DVC_ENABLE_DIRECTML)
  if(NOT WIN32)
    message(FATAL_ERROR "DirectML is supported only on Windows")
  endif()
  target_compile_definitions(dvc PRIVATE DVC_ENABLE_DIRECTML)
endif()
if(DVC_BUILD_EXAMPLES)
  add_executable(dvc_wav examples/wav.cpp)
  target_compile_features(dvc_wav PRIVATE cxx_std_17)
  target_link_libraries(dvc_wav PRIVATE dvc)
endif()
if(BUILD_TESTING)
  add_executable(dvc_api_test tests/api.c)
  target_link_libraries(dvc_api_test PRIVATE dvc)
  add_test(NAME api COMMAND dvc_api_test)
  add_executable(dvc_model_test tests/model.cpp)
  target_compile_features(dvc_model_test PRIVATE cxx_std_17)
  target_link_libraries(dvc_model_test PRIVATE dvc)
  set(DVC_TEST_MODELS "" CACHE PATH "Directory produced by tests/make_fixtures.py")
  if(DVC_TEST_MODELS)
    add_test(NAME model_contract COMMAND dvc_model_test "${DVC_TEST_MODELS}")
  endif()
endif()
# Runtime DLLs and third-party binaries are intentionally not bundled by install.
install(TARGETS dvc EXPORT dvcTargets RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES LICENSE LICENSE_STATUS.md THIRD_PARTY_NOTICES.md DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/dvc)
install(DIRECTORY licenses/ DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/dvc/third-party)
install(EXPORT dvcTargets NAMESPACE dvc:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/dvc)
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/dvcConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/dvcConfig.cmake" INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/dvc)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/dvcConfigVersion.cmake" VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/dvcConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/dvcConfigVersion.cmake" DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/dvc)
